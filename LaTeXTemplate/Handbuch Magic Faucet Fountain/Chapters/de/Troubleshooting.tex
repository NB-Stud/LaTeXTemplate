\chapter{Troubleshooting}
\begin{table}[h]
	\centering
	\caption{Troubleshooting des Magic Faucet Fountain}
	\renewcommand{\arraystretch}{1.4}
	\begin{tabular}{|>{\centering\arraybackslash}m{4.5cm}|
			>{\centering\arraybackslash}m{5.2cm}|
			>{\centering\arraybackslash}m{5cm}|}
		\hline
		\textbf{Problem} & \textbf{Mögliche Ursache} & \textbf{Fehlerbehebung} \\
		\hline
		\textbf{Kein Wasserfluss} 
		& \begin{enumerate}[nosep,leftmargin=*]
			\item Pumpe nicht angeschlossen
			\item Pumpe defekt
		\end{enumerate}
		& \begin{itemize}[nosep,leftmargin=*]
			\item Netzteil und Kabel überprüfen
			\item Ggf. Pumpe austauschen
		\end{itemize} \\
		\hline
		\textbf{Wasser läuft ungleichmäßig / schlechter Wasserkreislauf} 
		& \begin{enumerate}[nosep,leftmargin=*]
			\item Verschmutzung im Rohr
			\item Falsche Ausrichtung
			\item Rohr undicht
			\item Pumpe läuft unregelmäßig
		\end{enumerate}
		& \begin{itemize}[nosep,leftmargin=*]
			\item Rohr reinigen und neu montieren
			\item Ausrichtung kontrollieren
			\item Undichtigkeiten beheben
			\item Pumpe prüfen
		\end{itemize} \\
		\hline
		\textbf{Pumpe macht Geräusche} 
		& \begin{enumerate}[nosep,leftmargin=*]
			\item Verschmutzte Pumpe
			\item Luft oder Trockenlauf
		\end{enumerate}
		& \begin{itemize}[nosep,leftmargin=*]
			\item Pumpe reinigen oder tauschen
			\item Wasserstand erhöhen oder entlüften
		\end{itemize} \\
		\hline
		\textbf{Wasser läuft über} 
		& \begin{enumerate}[nosep,leftmargin=*]
			\item Zu viel Wasser
			\item Rücklauf blockiert
		\end{enumerate}
		& \begin{itemize}[nosep,leftmargin=*]
			\item Wasserstand reduzieren
			\item Rücklauf freimachen
		\end{itemize} \\
		\hline
		\textbf{Kein Strom} 
		& \begin{enumerate}[nosep,leftmargin=*]
			\item Kabelbruch
			\item Stromversorgung unterbrochen
		\end{enumerate}
		& \begin{itemize}[nosep,leftmargin=*]
			\item Leitungen prüfen
			\item Ggf. Netzteil tauschen
		\end{itemize} \\
		\hline
		\textbf{Keine Reaktion auf Eingaben} 
		& \begin{enumerate}[nosep,leftmargin=*]
			\item Bluetooth-Verbindung unterbrochen
			\item App eingefroren
			\item Arduino eingefroren
		\end{enumerate}
		& \begin{itemize}[nosep,leftmargin=*]
			\item Entfernung verringern
			\item App neustarten
			\item Arduino neustarten
		\end{itemize} \\
		\hline
	\end{tabular}
