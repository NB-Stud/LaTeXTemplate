	\caption{Troubleshooting des Magic Faucet Fountain}

		\hline
		\textbf{Problem} & \textbf{Mögliche Ursache} & \textbf{Fehlerbehebung} \\
		\hline
		\textbf{Kein Wasserfluss} 
		& \begin{enumerate}[nosep]
			\item Pumpe nicht angeschlossen
			\item Pumpe defekt
		\end{enumerate}
		& Netzteil und Kabel überprüfen \newline Ggf. Pumpe austauschen \\
		\hline
		\textbf{Wasser läuft ungleichmäßig / schlechter Wasserkreislauf} 
		& \begin{enumerate}[nosep]
			\item Verschmutzung im Rohr
			\item Falsche Ausrichtung
			\item Rohr undicht
			\item Pumpe läuft unregelmäßig
		\end{enumerate}
		& Rohr reinigen und neu montieren \newline Ausrichtung kontrollieren \newline Undichtigkeiten beheben \newline Pumpe prüfen \\
		\hline
		\textbf{Pumpe macht Geräusche} 
		& \begin{enumerate}[nosep]
			\item Verschmutzte Pumpe
			\item Luft oder Trockenlauf
		\end{enumerate}
		& Pumpe reinigen oder tauschen \newline Wasserstand erhöhen oder entlüften \\
		\hline
		\textbf{Wasser läuft über} 
		& \begin{enumerate}[nosep]
			\item Zu viel Wasser
			\item Rücklauf blockiert
		\end{enumerate}
		& Wasserstand reduzieren \newline Rücklauf freimachen \\
		\hline
		\textbf{Kein Strom} 
		& \begin{enumerate}[nosep]
			\item Kabelbruch
			\item Stromversorgung unterbrochen
		\end{enumerate}
		& Leitungen prüfen \newline Ggf. Netzteil tauschen \\
		\hline
		\textbf{Keine Reaktion auf Eingaben} 
		& \begin{enumerate}[nosep]
			\item Bluetooth-Verbindung unterbrochen
			\item App eingefroren
			\item Arduino eingefroren
		\end{enumerate}
		& Entfernung verringern \newline App neustarten \newline Arduino neustarten \\
		\hline
	\end{tabular}
\end{table}
