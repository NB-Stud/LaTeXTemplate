\chapter{Troubleshooting}
\begin{table}[h]
	\centering
	\caption{Troubleshooting des Magic Faucet Fountain}
	\begin{tabular}{p{4.5cm}|p{5cm}|p{5cm}}
		%\hline
		\textbf{Problem} & \textbf{Mögliche Ursache} & \textbf{Fehlerbehebung} \\
		\hline
		\textbf{Kein Wasserfluss} 
		& - Pumpe nicht angeschlossen \newline - Pumpe defekt 
		& - Netzteil und Kabel überprüfen \newline - Ggf. Pumpe austauschen \\
		\hline
		\textbf{Wasser läuft ungleichmäßig / schlechter Wasserkreislauf} 
		& - Verschmutzung oder Verstopfung im Rohr \newline 
		- Falsche Ausrichtung des Rohrs \newline 
		- Rohr undicht \newline 
		- Pumpe pumpt ungleichmäßig 
		& - Rohr demontieren, reinigen und erneut montieren \newline 
		- Rohr richtig zentrieren \newline 
		- Rohr auf undichte Stellen überprüfen \newline 
		- Pumpe auf gleichmäßigen Lauf prüfen \\
		\hline
		\textbf{Pumpe macht Geräusche} 
		& - Verschmutzte Pumpe \newline 
		- Pumpe läuft trocken oder hat Luft im System 
		& - Pumpe reinigen, ggf. ersetzen \newline 
		- Wasserstand im Tank erhöhen oder System entlüften \\
		\hline
		\textbf{Wasser läuft über} 
		& - Zu viel Wasser im System \newline 
		- Rücklauf blockiert 
		& - Wasserstand im Tank reduzieren \newline 
		- Rücklaufkanal frei machen \\
		\hline
		\textbf{Kein Strom} 
		& - Kabelbruch \newline 
		- Stromversorgung unterbrochen 
		& - Leitungen und Verbindungen überprüfen \newline 
		- Ggf. Leitungen und Netzteil austauschen \\
		\hline
		\textbf{Keine Reaktion auf Eingaben} 
		& - Bluetooth-Verbindung unterbrochen \newline 
		- App eingefroren \newline 
		- Arduino eingefroren 
		& - Entfernung reduzieren \newline 
		- App neustarten \newline 
		- Arduino neustarten \\
		\hline
	\end{tabular}
\end{table}
