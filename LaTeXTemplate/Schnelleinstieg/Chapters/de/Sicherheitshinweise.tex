\section{Sicherheitshinweise}
\begin{enumerate}
 \item Nur im Innenbereich verwenden.
\item Wasser regelmäßig nachfüllen, nicht trocken betreiben.
\item Maximalen Füllstand beachten.
\item Bluetooth-Verbindung vor Nutzung prüfen.
\item Netzteilkasten nicht öffnen!
\item Nur mit Originalnetzteil betreiben.
\item Elektronik trocken halten, keine Umbauten vornehmen.
\item Kinder nicht unbeaufsichtigt am Gerät spielen lassen.
\item Bei Störungen (Rauch, Geräusche, Geruch): sofort ausschalten und Netzstecker ziehen.
\end{enumerate}
